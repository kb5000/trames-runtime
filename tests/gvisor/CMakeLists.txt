# Small standalone build of unmodified upstream syscall tests. No runsc/Go/Bazel.
cmake_minimum_required(VERSION 3.20)
project(trames_gvisor_syscalls LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR NOT CMAKE_SYSTEM_PROCESSOR MATCHES "aarch64|arm64")
  message(FATAL_ERROR "Build on native ARM64 Linux for the reference/guest pair")
endif()
if(NOT GVISOR_SOURCE)
  message(FATAL_ERROR "Set GVISOR_SOURCE to the pinned test-only source directory")
endif()
foreach(dependency IN ITEMS BENCHMARK_SOURCE ABSL_SOURCE GTEST_SOURCE)
  if(NOT ${dependency})
    message(FATAL_ERROR "Set ${dependency} to its pinned dependency source")
  endif()
endforeach()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(BENCHMARK_ENABLE_TESTING OFF CACHE BOOL "" FORCE)
set(BENCHMARK_ENABLE_INSTALL OFF CACHE BOOL "" FORCE)
set(ABSL_BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(ABSL_PROPAGATE_CXX_STD ON CACHE BOOL "" FORCE)
set(INSTALL_GTEST OFF CACHE BOOL "" FORCE)
add_subdirectory("${ABSL_SOURCE}" absl-build)
add_subdirectory("${GTEST_SOURCE}" gtest-build)
add_subdirectory("${BENCHMARK_SOURCE}" benchmark-build)
add_compile_options(-O2 -g0 -march=armv8-a -ffunction-sections -fdata-sections -fno-sized-deallocation)
add_compile_definitions(_GNU_SOURCE)
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/include/tools/cpp/runfiles")
configure_file("${GVISOR_SOURCE}/tools/cpp/runfiles/runfiles_src.h" "${CMAKE_CURRENT_BINARY_DIR}/include/tools/cpp/runfiles/runfiles.h" COPYONLY)
include_directories("${CMAKE_CURRENT_BINARY_DIR}/include" "${GVISOR_SOURCE}" "${ABSL_SOURCE}" "${GTEST_SOURCE}/googletest/include" "${GTEST_SOURCE}/googlemock/include" "${BENCHMARK_SOURCE}/include")
if(GVISOR_EXPANDED_GRAPH)
  include("${GVISOR_EXPANDED_GRAPH}")
  return()
endif()
set(UTILS epoll_util fs_util logging posix_error save_util save_util_linux test_util test_util_impl test_util_runfiles temp_path multiprocess_util signal_util linux_capability_util socket_util time_util timer_util proc_util rlimit_util)
set(SUPPORT "${GVISOR_SOURCE}/tools/cpp/runfiles/runfiles_src.cc")
foreach(name IN LISTS UTILS)
  list(APPEND SUPPORT "${GVISOR_SOURCE}/test/util/${name}.cc")
endforeach()
add_library(gvisor_support OBJECT ${SUPPORT})
set(SUITES brk clock_getres clock_gettime dup eventfd getrandom pipe epoll mmap futex uname)
foreach(suite IN LISTS SUITES)
  add_executable(${suite}_test "${GVISOR_SOURCE}/test/syscalls/linux/${suite}.cc"
    "${GVISOR_SOURCE}/test/util/test_main.cc"
    "${GVISOR_SOURCE}/test/util/errno_safe_allocator.cc"
    $<TARGET_OBJECTS:gvisor_support>)
  target_link_options(${suite}_test PRIVATE -static -Wl,--gc-sections)
  target_link_libraries(${suite}_test PRIVATE GTest::gmock GTest::gtest benchmark absl::flags_parse absl::str_format absl::strings absl::synchronization absl::time pthread dl rt)
endforeach()
